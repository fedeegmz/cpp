## Trazas

**Traza 1**

```cpp
#include<iostream>
using namespace std;

int main(){
    int contUno=1, contDos=2, aux;

    while(contUno < 8){
        aux = contDos - contUno;
        if (aux % 4 == 0){
            cout<<"A\n";
        } else cout<<"B\n";
        contUno++;
        contDos+=2;
    }
}
```

**Traza 2**

```cpp
#include<iostream>
using namespace std;

int main(){
    int cont=10, suma=0;
    do{
        switch (cont){
            case 10: case 2: case 4:
                suma+=5;
                break;

            case 9: case 8: case 5:
                suma-=10;
                break;

            default:
                suma+=15;
        }
        cont--;
    } while(cont != 2);
    cout<<"La suma es "<<suma;
}
```
